function formatNum(n: string, len: number) {
  const s = n.padStart(len, "0");
  return s.slice(-len);
}

  const {

    [games, gameId]
  );
  const activeBet = useMemo(

    [betTypes, bt]
  );

    if (first && !bt) setBt(first.key);

    activeBet?.mode === "4cifra"

      : activeBet?.mode === "3cifra"

          : activeBet?.mode === "skina" || activeBet?.mode === "bf"

    if (k === "CLR") {
      if (field === "num") setNum("");
      else setAmt(1);
      return;

    if (k === "BS") {
      if (field === "num") setNum((n) => n.slice(0, -1));
      else setAmt((a) => Math.floor(a / 10) || 1);
      return;
    }
    if (field === "num") {
      if (num.length >= digitLen) return;
      setNum((n) => n + k);
    } else {
      const next = Number(String(amt) + k);
      if (next <= 9999) setAmt(next);
    }
  }

  function quickPick() {
    if (!activeBet) return;
    const max = 10 ** digitLen - 1;
    const r = Math.floor(Math.random() * (max + 1));
    setNum(formatNum(String(r), digitLen));
  }

    if (!lastTicket?.lines?.[0]) return;
    const L = lastTicket.lines[0];

    setBt(L.betTypeKey);
    setNum(L.number ?? "");
    setAmt(L.amount ?? 1);
    setMultiDraw(L.multiDraw ?? 1);
  }

        number:

        amount: amt,
        multiDraw,
      });
      setJackB([]);
      setJackMega(null);
      return;
    }

    addLine({
      gameId: game.id,
      betTypeKey: activeBet.key,
      number: num || undefined,
      amount: amt,
      multiDraw,
    });
    setNum("");
  }
